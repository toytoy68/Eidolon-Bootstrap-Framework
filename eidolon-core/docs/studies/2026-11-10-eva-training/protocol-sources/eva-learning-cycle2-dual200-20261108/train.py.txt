progress('importing')

exec(compile(ast.Module(body=nodes,type_ignores=[]),str(old),'exec'))
data=json.loads((ROOT/'dataset.json').read_text())
assert hashlib.sha256((ROOT/'dataset.json').read_bytes()).hexdigest()==json.loads((ROOT/'lineage.json').read_text())['dataset_sha256']
train=data['train'];valid=data['validation'];heldout=data['heldout']
train_batches,train_info=encode(train);valid_batches,valid_info=encode(valid)
save('dataset-statistics-rank'+str(RANK)+'.json',dict(train=train_info,validation=valid_info))
progress('loading_model')

model=AutoModelForCausalLM.from_pretrained(str(ROOT/'source-model'),local_files_only=True,trust_remote_code=False,quantization_config=quant,torch_dtype=dtype,device_map={'':RANK},attn_implementation='sdpa')
model.config.use_cache=False
model=prepare_model_for_kbit_training(model,use_gradient_checkpointing=True,gradient_checkpointing_kwargs={'use_reentrant':False})
model=PeftModel.from_pretrained(model,C['parent_adapter'],is_trainable=True)
params=[p for p in model.parameters() if p.requires_grad]
assert sum(p.numel() for p in params)==16515072
assert all(p.dtype==torch.float32 for p in params)
def statehash():
 return hashlib.sha256(b''.join(v.detach().cpu().contiguous().view(torch.uint8).numpy().tobytes() for k,v in sorted(get_peft_model_state_dict(model).items()))).hexdigest()
initial_hash=statehash();hashes=[None]*WORLD;dist.all_gather_object(hashes,initial_hash)
assert len(set(hashes))==1,'Different initial adapters'
save('qualification-rank'+str(RANK)+'.json',dict(rank=RANK,gpu=torch.cuda.get_device_name(RANK),uuid=C['uuids'][RANK],precision=str(dtype),trainable_parameters=sum(p.numel() for p in params),initial_adapter_state_sha256=initial_hash,device_map={k:str(v) for k,v in model.hf_device_map.items()},torch=torch.__version__,attention='sdpa',quantization='NF4 double quant',mode=MODE))
progress('ddp_setup')
ddp=DDP(model,device_ids=[RANK],output_device=RANK,broadcast_buffers=False,find_unused_parameters=False)
def batch(b):return {k:v.to('cuda',non_blocking=True) for k,v in b.items()}
def barrier():
 progress('waiting_peer')
 dist.barrier(device_ids=[RANK])
def loss_evaluation():
 model.eval();losses=[]
 with torch.no_grad():
  for b in valid_batches[:8]:
   with torch.autocast('cuda',dtype=dtype):loss=model(**batch(b)).loss
   assert torch.isfinite(loss)
   losses.append(float(loss))
 model.train();return sum(losses)/len(losses)
def generation(stage):
 model.eval();rows=[];start=time.time()
 local=heldout[RANK::WORLD];canaries=CANARIES[RANK::WORLD]
 progress('evaluation_'+stage,done=0,total=len(local)+len(canaries))
 for i,ex in enumerate(local):
  ids=tokenizer.apply_chat_template([dict(role='system',content=SYSTEM),dict(role='user',content=ex['question'])],tokenize=True,add_generation_prompt=True,enable_thinking=False,return_tensors='pt').to('cuda')
  with torch.inference_mode(),torch.autocast('cuda',dtype=dtype):
   out=model.generate(input_ids=ids,attention_mask=torch.ones_like(ids),max_new_tokens=192,do_sample=False,use_cache=True,pad_token_id=tokenizer.pad_token_id,eos_token_id=tokenizer.eos_token_id)
  text=tokenizer.decode(out[0,ids.shape[1]:],skip_special_tokens=True)
  r=dict(stage=stage,rank=RANK,id=ex['id'],family=ex['family'],question=ex['question'],expected=ex['expected'],response=text,evaluation=grade(text,ex['expected']),generated_tokens=int(out.shape[1]-ids.shape[1]),token_limit_reached=out.shape[1]-ids.shape[1]>=192)
  rows.append(r);append('evaluation-rank'+str(RANK)+'.jsonl',r);progress('evaluation_'+stage,done=i+1,total=len(local)+len(canaries))
 for j,(name,q) in enumerate(canaries):
  ids=tokenizer.apply_chat_template([dict(role='system',content='Tu es un assistant francophone. Réponds à la demande sans inventer les informations manquantes.'),dict(role='user',content=q)],tokenize=True,add_generation_prompt=True,enable_thinking=False,return_tensors='pt').to('cuda')
  with torch.inference_mode(),torch.autocast('cuda',dtype=dtype):
   out=model.generate(input_ids=ids,attention_mask=torch.ones_like(ids),max_new_tokens=256,do_sample=False,use_cache=True,pad_token_id=tokenizer.pad_token_id,eos_token_id=tokenizer.eos_token_id)
  append('canaries-rank'+str(RANK)+'.jsonl',dict(stage=stage,rank=RANK,name=name,question=q,response=tokenizer.decode(out[0,ids.shape[1]:],skip_special_tokens=True),token_limit_reached=out.shape[1]-ids.shape[1]>=256))
  progress('evaluation_'+stage,done=len(local)+j+1,total=len(local)+len(canaries))
 save('evaluation-'+stage+'-rank'+str(RANK)+'.json',dict(rows=rows,seconds=time.time()-start))
 barrier()
 if RANK==0:
  allrows=sum([json.loads((P/('evaluation-'+stage+'-rank'+str(r)+'.json')).read_text())['rows'] for r in range(WORLD)],[])
  assert len(allrows)==160 and len({r['id'] for r in allrows})==160
  save('evaluation-'+stage+'.json',dict(n=160,content_correct=sum(r['evaluation']['content_correct'] for r in allrows),exact_json=sum(r['evaluation']['exact'] for r in allrows),families={str(f):dict(n=sum(r['family']==f for r in allrows),correct=sum(r['family']==f and r['evaluation']['content_correct'] for r in allrows)) for f in range(8)},seconds=time.time()-start,precision='FP16',evaluation_distribution='fixed round-robin questions across two GPUs'))
 barrier();model.train()

def agreement():
 flat=torch.cat([p.detach().flatten() for p in params]);ref=flat.clone();dist.broadcast(ref,src=0)
 delta=(flat-ref).abs().max();dist.all_reduce(delta,op=dist.ReduceOp.MAX)
 value=float(delta);assert value<=1e-6,'Adapter replicas diverged: '+str(value)
 return value
def update(optimizer,scaler,index):
 optimizer.zero_grad(set_to_none=True);start=time.time()
 indices=list(range(index*4,min(index*4+4,len(train_batches))))
 total_supervised=sum(int((train_batches[k]['labels'][:,1:]!=-100).sum()) for k in indices)
 assert total_supervised>0
 nonpad=supervised=padded=0;losses=[];local_indices=[]
 for j in range(2):
  k=index*4+j*2+RANK;real=k<len(train_batches)
  cpu=train_batches[k if real else indices[-1]]
  n=int((cpu['labels'][:,1:]!=-100).sum()) if real else 0
  if real:
   nonpad+=int(cpu['attention_mask'].sum());supervised+=n;padded+=cpu['input_ids'].numel();local_indices.append(k)
  with ddp.no_sync() if j==0 else contextlib.nullcontext():
   with torch.autocast('cuda',dtype=dtype):loss=ddp(**batch(cpu)).loss
   if not torch.isfinite(loss):raise RuntimeError('Nonfinite loss')
   if real:losses.append(float(loss.detach()))
   scaler.scale(loss*(WORLD*n/total_supervised)).backward()
 scaler.unscale_(optimizer);norm=torch.nn.utils.clip_grad_norm_(params,1.0)
 if not torch.isfinite(norm):raise RuntimeError('Nonfinite gradient norm')
 old=scaler.get_scale();scaler.step(optimizer);scaler.update();torch.cuda.synchronize()
 if scaler.get_scale()<old:raise RuntimeError('Optimizer overflow')
 return dict(rank=RANK,seconds=time.time()-start,pack_indices=local_indices,nonpadding_tokens=nonpad,supervised_tokens=supervised,padded_tokens=padded,loss=sum(losses)/len(losses) if losses else None,gradient_norm=float(norm),lr=optimizer.param_groups[0]['lr'],scale=scaler.get_scale(),peak_vram_mib=torch.cuda.max_memory_allocated()/1048576)
progress('loaded')
barrier()
if MODE=='train':
 baseline=loss_evaluation();save('baseline-loss-rank'+str(RANK)+'.json',dict(loss=baseline,scope='first 8 validation packs only'))
 generation('before')
else:
 baseline=loss_evaluation();save('baseline-loss-rank'+str(RANK)+'.json',dict(loss=baseline))
optimizer=torch.optim.AdamW(params,lr=5e-5,weight_decay=0.0,foreach=False)
scaler=torch.amp.GradScaler('cuda',init_scale=128.0,growth_interval=2000,enabled=True)
full_steps=math.ceil(len(train_batches)/4)
steps=4 if MODE=='smoke' else full_steps
warm=max(1,int(full_steps*.05))
scheduler=torch.optim.lr_scheduler.LambdaLR(optimizer,lambda i:min((i+1)/warm,1.0)*max(.1,1-i/full_steps))
torch.cuda.empty_cache();torch.cuda.reset_peak_memory_stats();start=time.time()
rows=[];progress('training',optimizer_step=0,total_steps=steps)
for i in range(steps):
 r=update(optimizer,scaler,full_steps-1 if MODE=='smoke' and i==3 else i);r.update(optimizer_step=i+1,epoch=time.time());rows.append(r)

 progress('training',optimizer_step=i+1,total_steps=steps)
 if MODE=='train' and (i+1)%100==0 and RANK==0:
  progress('checkpoint',optimizer_step=i+1,total_steps=steps);model.save_pretrained(P/'checkpoints'/('step-'+str(i+1)))
delta=agreement();final_hash=statehash();assert final_hash!=initial_hash,'No adapter learning'
save('training-rank'+str(RANK)+'.json',dict(mode=MODE,seconds=time.time()-start,optimizer_steps=steps,nonpadding_tokens=sum(r['nonpadding_tokens'] for r in rows),supervised_tokens=sum(r['supervised_tokens'] for r in rows),initial_adapter_state_sha256=initial_hash,final_adapter_state_sha256=final_hash,replica_max_difference=delta,peak_vram_mib=torch.cuda.max_memory_allocated()/1048576))
barrier()
if MODE=='smoke':
 if RANK==0:save('passed.json',dict(passed=True,steps=steps,replica_max_difference=delta,epoch=time.time(),parent_unmodified=True))
else:
 if RANK==0:
  counts=[json.loads((P/('training-rank'+str(r)+'.json')).read_text()) for r in range(WORLD)]
  assert sum(r['nonpadding_tokens'] for r in counts)==train_info['tokens'],'Coverage mismatch'

  progress('saving');model.save_pretrained(P/'adapter-final');tokenizer.save_pretrained(P/'adapter-final')
