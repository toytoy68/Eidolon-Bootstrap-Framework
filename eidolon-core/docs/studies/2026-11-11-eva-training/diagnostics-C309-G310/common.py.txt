from pathlib import Path
import json,os,time,hashlib,re
P=Path(__file__).resolve().parent
def read(path):return json.loads(Path(path).read_text())
def sha(path):
 h=hashlib.sha256()
 with Path(path).open('rb') as f:
  for b in iter(lambda:f.read(8388608),b''):h.update(b)
 return h.hexdigest()
def save(path,obj):
 path=Path(path);path.parent.mkdir(exist_ok=True,parents=True);q=path.with_name(path.name+'.'+str(os.getpid())+'.tmp')
 with q.open('w') as f:json.dump(obj,f,ensure_ascii=False);f.flush();os.fsync(f.fileno())
 os.replace(q,path)
 fd=os.open(path.parent,os.O_DIRECTORY)
 try:os.fsync(fd)
 finally:os.close(fd)
def item_path(out,stage,candidate,ident):
 assert re.fullmatch(r'[A-Za-z0-9+_.-]+',ident),ident
 return Path(out)/'items'/(stage+'--'+candidate+'--'+ident+'.json')
