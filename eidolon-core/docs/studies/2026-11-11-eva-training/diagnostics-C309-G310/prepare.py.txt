seen=set()
def collect(obj):
 if isinstance(obj,dict):
  if isinstance(obj.get('question'),str):
   op=time_operands(obj['question'])
   if op:seen.add(tuple(op))
   if obj.get('time_key'):seen.add(tuple(obj['time_key']))
  else:
   for x in obj.values():collect(x)
 elif isinstance(obj,list):
  for x in obj:collect(x)
for root in P.parent.glob('eva-learning-*'):
 f=root/'dataset.json'
 if f.is_file():collect(read(f))
