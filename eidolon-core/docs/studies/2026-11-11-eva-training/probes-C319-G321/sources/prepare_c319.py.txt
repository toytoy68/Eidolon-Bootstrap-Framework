from common import P,read,save,sha
import re,json,collections,importlib.util,ast
import eva_boundary_probes as bp

assert max(lengths)<2048

 tree=ast.parse((P/name).read_text());calls=[x.func.attr if isinstance(x.func,ast.Attribute) else x.func.id if isinstance(x.func,ast.Name) else '' for x in ast.walk(tree) if isinstance(x,ast.Call)]
 assert not set(calls)&{'backward','step','train','init_process_group','DistributedDataParallel'},(name,set(calls))
