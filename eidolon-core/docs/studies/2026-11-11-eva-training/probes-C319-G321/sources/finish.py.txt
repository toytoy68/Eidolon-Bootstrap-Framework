rows=collect('sign')
for k in ['ZS','FA','FB','FC']:

for a,b in [('FC','FA'),('FC','FB'),('FC','ZS'),('FA','ZS'),('FB','ZS')]:

 gains=sum(v[a]['correct'] and not v[b]['correct'] for v in ps);losses=sum(v[b]['correct'] and not v[a]['correct'] for v in ps);n=gains+losses
 comparisons[a+'__'+b]=dict(n=len(ps),gains=gains,losses=losses,p_exact_exploratory=min(1.,2*sum(math.comb(n,i) for i in range(min(gains,losses)+1))/2**n) if n else 1.)

assert not (OUT/'guard-trip.json').exists()
result=dict(state='finished',training=False,planned=200,generated=len(rows),stats=stats,paired=comparisons,claude_grade=bp.grade(planned,rows),strict_sign_gate=gate,telemetry=telemetry,plan=read(P/'experiment-plan.json'),FC_at_least_FA_content=stats['FC']['correct']>=stats['FA']['correct'],automatic_training=False,automatic_promotion=False,limits=['Known44cases and shot pairs reused, fresh generations at batch16; no claimed equivalence to historical batch1','Counter regex checks written literal sequence only, not complete algorithm','Validator partial, decimal claims intentionally outside scope','Showing examples in context is not proof that training C prevents a hybrid transition'])
