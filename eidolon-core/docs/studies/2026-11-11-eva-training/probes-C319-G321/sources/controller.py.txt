def notify(ready=False):

 if a:
  if a.startswith('@'):a='\0'+a[1:]
  with socket.socket(socket.AF_UNIX,socket.SOCK_DGRAM) as s:s.connect(a);s.sendall((b'READY=1\n' if ready else b'')+b'WATCHDOG=1')

 d=dict(phase=phase,heartbeat=time.time(),epoch=time.time(),training=False,**kw);save(OUT/'status.json',d);save(P/'campaign-status.json',d);notify()

def collect(stage):return sum([read(OUT/f'{stage}-parent-rank{r}.json')['rows'] for r in range(2)],[])
def strict_sign(r,p):
 import eva_boundary_probes as bp
 o=bp._last_object(r['response']) or {};v=o.get('resultat')
 if isinstance(v,str):
  import re
  if re.fullmatch(r'\s*[−-]?\d+\s*',v):v=int(v.strip().replace('−','-'))
 return isinstance(v,int) and not isinstance(v,bool) and v==p['expected']['resultat'] and o.get('signe')==p['expected']['signe'] and o.get('unite')=='nombre'
def gate():
 ps={x['id']:x for x in read(P/'boundary-probes.json') if x['kind']=='SG'};rows=collect('sign');assert len(rows)==24 and {x['id'] for x in rows}==ps.keys()
 wrong=[x['id'] for x in rows if not strict_sign(x,ps[x['id']])]
 save(OUT/'sign-gate.json',dict(allow_B=not wrong,wrong=wrong,n=24,correct=24-len(wrong),bool_rejected=True,training=False))

 deadline=time.time()+timeout

   f=OUT/f'progress-rank{r}.json'
   if f.exists():
    d=read(f);assert d['stage']=='finished' or time.time()-d['epoch']<600,'Progress stale'
  if phase=='probes' and not (OUT/'sign-gate.json').exists() and all((OUT/f'sign-ready-rank{r}.json').exists() for r in range(2)):gate()

 children=[sp.Popen([str(P/'venv/bin/python'),str(P/'worker.py')],env=dict(env,EVA_RANK=str(r),EVA_DIAGNOSTIC='C319-probes',CUDA_VISIBLE_DEVICES=c['uuids'][r]),stdout=(OUT/f'worker-rank{r}.log').open('w'),stderr=sp.STDOUT,start_new_session=True) for r in range(2)]
