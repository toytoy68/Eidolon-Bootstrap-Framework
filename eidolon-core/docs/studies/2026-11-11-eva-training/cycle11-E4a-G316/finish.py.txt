def rows(folder,stage,candidate):
 out=[]
 for rank in range(2):out+=read(P/folder/f'{stage}-{candidate}-rank{rank}.json')['rows']

 return out
def score(rr):
 return dict(n=len(rr),correct=sum(x['evaluation']['content_correct'] for x in rr),truncated=sum(x['truncated'] for x in rr),starts_Depart=sum(x.get('chain_metrics',{}).get('starts_Depart',False) for x in rr),required_A_verified=sum(x.get('chain_metrics',{}).get('required_chain_A_verified',False) for x in rr),hybrid_prefix=sum(x.get('chain_metrics',{}).get('hybrid_prefix_without_required_chain',False) for x in rr),control_compares=sum(x.get('chain_metrics',{}).get('control_compare_count',0) for x in rr),control_counters=sum(x.get('chain_metrics',{}).get('control_counter_count',0) for x in rr),detected_false_claims=sum(bool(x.get('invalid_claims')) for x in rr))
def compare(a,b):
 aa={x['id']:x for x in a};bb={x['id']:x for x in b};assert aa.keys()==bb.keys()
 assert all(aa[k]['identity']['gpu_uuid']==bb[k]['identity']['gpu_uuid'] and aa[k]['identity']['batch_ids']==bb[k]['identity']['batch_ids'] for k in aa)
 g=sum(not aa[k]['evaluation']['content_correct'] and bb[k]['evaluation']['content_correct'] for k in aa);l=sum(aa[k]['evaluation']['content_correct'] and not bb[k]['evaluation']['content_correct'] for k in aa);n=g+l
 return dict(before=score(a),after=score(b),gains=g,losses=l,p_exact_exploratory=min(1.,2*sum(math.comb(n,i) for i in range(min(g,l)+1))/2**n) if n else 1.)
